void er_to_shi(int n){
	int a=0;
	int res=0;
	while(n!=0){
	
	
	int i=n%10;
	n/=10;
	res += i*pow(2,a);
	a++;
	}
	printf("%d\n",res);
}
void shi_to_er(int n){

	shi_to_er(n/2);
	
	}
