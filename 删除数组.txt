void valuedelete(int arr[],int arr1[],int n,int len){
	int flag =0;
	int j = 0;
	for(int i=0;i<len;i++){
		if(arr[i]!=n){
			arr1[j] = arr[i];
			j++;
		}else{
			flag = 1;
		}
	}
	if(!flag)printf("error");
}
void indexdelete(int arr[],int arr1[],int indes,int len){
	int index = indes-1;
	for(int j=0;j<index;j++){
		arr1[j] = arr[j];
	}
	for(int k=index;k<len-1;k++){
		arr1[k] = arr[k+1]; 
	}
}
