#include<stdio.h>
void shi_to_binary(int n,int arr[]);

int main()
{
	int arr[32];
	int n;
	scanf("%d",&n);
	shi_to_binary(n,arr);
	return 0;
}
void shi_to_binary(int n,int arr[]){
	int i=0;
	while(n>0){
		arr[i]=n%2;
		n/=2;
	 
		i++;
	}
	for(i=i-1;0<=i;i--){
		printf("%d",arr[i]);
	}
	
}
