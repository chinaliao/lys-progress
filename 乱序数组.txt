#include<stdio.h>
#include<stdlib.h>
#include<time.h>


int main()
{
	
	int arr[]={1,2,3,4,5};
	int len=sizeof(arr)/sizeof(int);
	srand(time(NULL));
	int j=rand()%len;
	for(int i=0;i<len;i++){
		int a=arr[i];
		arr[i]=arr[j];
		arr[j]=a;
	}
	for(int i=0;i<len;i++){
		printf("%d\n",arr[i]);
	}
	
    
	
	return 0;
}
