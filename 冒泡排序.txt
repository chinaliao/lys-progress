#include<stdio.h>
int main()
{
  int	arr[10]={1,2,4,6,3,5,8,7,9,0};
	int len=sizeof(arr)/sizeof(int);
	for(int i=0;i<len-1;i++){
		for(int j=0;j<len-1-i;j++)
		{
		if(arr[j]>arr[j+1]){
			int temp=arr[j];
			arr[j]=arr[j+1];
			arr[j+1]=temp;
		}
			
		}   
	}

	for(int i=0;i<len;i++){
		printf("%d",arr[i]);//TODO
	}
	
	return 0;
}
